\section{符号说明}

\begin{table}[H]
\centering
\caption{主要符号及含义}
\label{tab:symbols}
\begin{tabularx}{0.96\textwidth}{cX}
\toprule
符号 & 含义\\
\midrule
$x_i^{(e)}(t)$ & 第$i$个试次、电极$e$在时刻$t$的原始脑电观测\\
$f_s$ & EEG采样频率，本文数据为$256\,\mathrm{Hz}$\\
$\mathcal{T}_0$ & 刺激前基线时间窗\\
$x_i^{(0)}(t)$ & 基础保真处理$\Pzero$后的第$i$个试次信号\\
$w_i$ & 第$i$个试次的质量权重\\
$N_{\mathrm{eff}}$ & 加权后的有效样本量\\
$U$ & 候选视觉响应子空间的正交基\\
$P=UU^\top$ & 候选响应子空间投影矩阵\\
$D_2$ & 二阶离散差分算子\\
$S_\lambda$ & 正则化平滑算子，$\lambda$为平滑强度\\
$\bar{x}_{c}^{(e)}(t)$ & 条件$c$下电极$e$的平均事件相关响应\\
$m(t)$ & 左、右条件的共同响应\\
$d(t)$ & 左、右条件的差分响应\\
$f_K(t)$ & 含$K$个高斯成分的参数化ERP曲线\\
$\bm u(t)$ & 问题二中由视觉刺激/LGN形成的外部输入\\
$\bm z(t)$ & 问题二中皮层神经群体的低维等效状态\\
$H$ & 隐状态到F3、Fz、F4头皮观测的映射矩阵\\
$\bm r(t)$ & 问题二输出的视觉刺激响应特征表示\\
$\bm s(t)$ & 问题三中记忆/认知相关低维状态\\
$\bm\theta,\bm\phi$ & 问题二、三模型的待估参数集合\\
\bottomrule
\end{tabularx}
\end{table}
